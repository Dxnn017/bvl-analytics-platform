=== DETECTOR GOLD ===
"""Detección de cambios efectivos entre dos versiones Gold.

Solo calcula diferencias en memoria. No escribe ni publica datos.
"""

from pyspark.sql import functions as F


CLAVES_PRODUCTO = {
    "mercado_valor_diario": ["nemonico", "fecha"],
    "actividad_sab_diaria": ["fecha", "codigo_sab"],
    "empresa_trimestre": ["id_registro"],
    "hechos_empresa": ["numero_expediente"],
    "impacto_hechos_valor": ["numero_expediente", "nemonico"],
}


def detectar_cambios_gold(anterior, vigente, producto):
    """Devuelve las claves anteriores y nuevas con cambios efectivos."""

    if producto not in CLAVES_PRODUCTO:
        raise ValueError(f"Producto Gold desconocido: {producto}")

    claves = CLAVES_PRODUCTO[producto]

    if set(anterior.columns) != set(vigente.columns):
        raise ValueError("Los esquemas Gold son incompatibles.")

    for nombre, datos in (
        ("anterior", anterior),
        ("vigente", vigente),
    ):
        invalidos = datos.filter(
            F.expr(
                " OR ".join(
                    f"`{clave}` IS NULL"
                    for clave in claves
                )
            )
        ).limit(1).count()

        if invalidos:
            raise ValueError(
                f"Existen claves nulas en Gold {nombre}."
            )

        duplicados = (
            datos.groupBy(*claves)
            .count()
            .filter(F.col("count") > 1)
            .limit(1)
            .count()
        )

        if duplicados:
            raise ValueError(
                f"Existen claves duplicadas en Gold {nombre}."
            )

    columnas = sorted(anterior.columns)

    antes = anterior.select(*columnas)
    despues = vigente.select(*columnas)

    filas_eliminadas_o_modificadas = antes.exceptAll(
        despues
    )

    filas_insertadas_o_modificadas = despues.exceptAll(
        antes
    )

    claves_anteriores = (
        filas_eliminadas_o_modificadas
        .select(*claves)
        .distinct()
    )

    claves_nuevas = (
        filas_insertadas_o_modificadas
        .select(*claves)
        .distinct()
    )

    return {
        "producto_gold": producto,
        "claves_anteriores": claves_anteriores,
        "claves_nuevas": claves_nuevas,
        "claves_afectadas": (
            claves_anteriores
            .unionByName(claves_nuevas)
            .distinct()
        ),
    }

=== COMPARADOR DE NEGOCIO ===
"""Comparación de cambios de negocio Gold. Propuesta local."""

import json
from pathlib import Path

BASE = Path("docs/bigquery")

# Exclusiones técnicas candidatas; sujetas a revisión de Daniela.
METADATOS_TECNICOS = {
    "fecha_ingesta",
    "operation",
    "delete_authorized",
    "run_id",
    "source_version",
    "batch_id",
    "payload_sha256",
}


def columnas_negocio(producto):
    contrato = json.loads(
        (BASE / f"{producto}.json").read_text()
    )

    columnas = [
        c["name"]
        for c in contrato["columns"]
        if c["name"] not in METADATOS_TECNICOS
    ]

    if not columnas:
        raise ValueError("Sin columnas de negocio")

    return columnas


def cambio_negocio(producto, anterior, nuevo):
    return any(
        anterior.get(columna) != nuevo.get(columna)
        for columna in columnas_negocio(producto)
    )


if __name__ == "__main__":
    casos = {
        "mercado_valor_diario": {
            "nemonico": "ABC",
            "fecha": "2026-11-27",
            "precio_cierre": 10.0,
        },
        "actividad_sab_diaria": {
            "fecha": "2026-11-27",
            "codigo_sab": "S01",
            "monto_total": 100.0,
        },
        "empresa_trimestre": {
            "id_registro": "REG-001",
            "activo_total": 1000.0,
        },
    }

    for producto, anterior in casos.items():
        solo_metadata = {
            **anterior,
            "fecha_ingesta": "2026-11-28",
        }

        assert not cambio_negocio(
            producto, anterior, solo_metadata
        )

        modificada = dict(anterior)

        campo = {
            "mercado_valor_diario": "precio_cierre",
            "actividad_sab_diaria": "monto_total",
            "empresa_trimestre": "activo_total",
        }[producto]

        modificada[campo] += 1

        assert cambio_negocio(
            producto, anterior, modificada
        )

        print(f"{producto}: COMPARADOR_NEGOCIO_OK")

    print("TRES_COMPARADORES_NEGOCIO_VALIDADOS_OK")

=== HASH PAYLOAD ===
"""Verificación SHA-256 local. No valida Parquet oficial."""

import hashlib
import json
import re


def calcular_sha256(filas):
    if not isinstance(filas, list) or not filas:
        raise ValueError("Payload vacío o inválido")

    contenido = json.dumps(
        filas,
        sort_keys=True,
        ensure_ascii=False,
        separators=(",", ":"),
        default=str,
    )
    return hashlib.sha256(contenido.encode("utf-8")).hexdigest()


def verificar_sha256(filas, hash_declarado):
    if not isinstance(hash_declarado, str):
        raise ValueError("SHA-256 inválido")

    if not re.fullmatch(r"[0-9a-fA-F]{64}", hash_declarado):
        raise ValueError("Formato SHA-256 inválido")

    calculado = calcular_sha256(filas)

    if calculado.lower() != hash_declarado.lower():
        raise ValueError("SHA-256 no coincide con el contenido")

    return True


if __name__ == "__main__":
    casos = {
        "mercado_valor_diario": {
            "nemonico": "ABC", "fecha": "2026-11-27"
        },
        "actividad_sab_diaria": {
            "fecha": "2026-11-27", "codigo_sab": "S01"
        },
        "empresa_trimestre": {
            "id_registro": "REG-001"
        },
    }

    for producto, clave in casos.items():
        filas = [{**clave, "operation": "UPSERT", "valor": 100}]
        huella = calcular_sha256(filas)

        assert verificar_sha256(filas, huella)

        alteradas = [{**clave, "operation": "UPSERT", "valor": 999}]

        try:
            verificar_sha256(alteradas, huella)
        except ValueError:
            pass
        else:
            raise AssertionError("Payload alterado no rechazado")

        print(f"{producto}: SHA256_CONTENIDO_VALIDADO_OK")

    print("TRES_VERIFICACIONES_SHA256_LOCALES_OK")
