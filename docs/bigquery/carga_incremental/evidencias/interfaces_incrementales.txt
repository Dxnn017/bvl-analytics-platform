-- PROPUESTA LOCAL. NO EJECUTAR SIN QUALITY.
CREATE TABLE `gold_analytics.actividad_sab_diaria` (
  `fecha` DATE,
  `codigo_sab` STRING,
  `sociedad_agente_bolsa` STRING,
  `monto_compras` NUMERIC,
  `monto_ventas` NUMERIC,
  `monto_total` NUMERIC,
  `porcentaje_compras_original` FLOAT64,
  `porcentaje_ventas_original` FLOAT64,
  `porcentaje_total_original` FLOAT64,
  `fuente_datos` STRING,
  `fecha_ingesta` DATE,
  `montos_validos` BOOL NOT NULL,
  `diferencia_monetaria` NUMERIC,
  `total_consistente` BOOL NOT NULL,
  `saldo_neto` NUMERIC,
  `total_compras_diario` NUMERIC,
  `total_ventas_diario` NUMERIC,
  `total_actividad_diaria` NUMERIC,
  `cantidad_sab_diaria` INT64 NOT NULL,
  `registros_invalidos_diarios` INT64,
  `actividad_diaria_valida` BOOL,
  `participacion_compras_pct` FLOAT64,
  `participacion_ventas_pct` FLOAT64,
  `participacion_total_pct` FLOAT64,
  `diferencia_participacion_compras` FLOAT64,
  `diferencia_participacion_ventas` FLOAT64,
  `diferencia_participacion_total` FLOAT64,
  `ranking_diario` INT64,
  `aporte_hhi` FLOAT64,
  `hhi_diario` FLOAT64,
  `anio` INT64,
  `mes` INT64,
  `trimestre` INT64,
  `registros_elegibles` INT64,
  `registros_cuarentena` INT64,
  `archivos_excluidos` INT64,
  `estado_cobertura` STRING NOT NULL,
  `version_silver` STRING,
  `scope_quality_id` STRING,
  `trazabilidad_verificada` BOOL NOT NULL,
  `cobertura_completa` BOOL NOT NULL
)
PARTITION BY `fecha`
CLUSTER BY `codigo_sab`;

    WHERE batch_id = @batch_id
      AND run_id = @run_id
      AND source_version = @source_version
      AND payload_sha256 = @payload_sha256
  )
  WHERE operation NOT IN ('UPSERT', 'DELETE')
     OR operation IS NULL
     OR run_id IS NULL
     OR source_version IS NULL
     OR (operation = 'DELETE'
         AND delete_authorized IS NOT TRUE)
) AS 'Operación inválida o eliminación no autorizada';

-- Validación de campos obligatorios para UPSERT.
ASSERT (
  SELECT COUNT(*) = 0
  FROM (
    SELECT *
    FROM `gold_analytics_staging.actividad_sab_diaria_delta`
    WHERE batch_id = @batch_id
      AND run_id = @run_id
      AND source_version = @source_version
      AND payload_sha256 = @payload_sha256
  )
  WHERE operation = 'UPSERT'
    AND (
      `montos_validos` IS NULL OR
      `total_consistente` IS NULL OR
      `cantidad_sab_diaria` IS NULL OR
       `estado_cobertura` IS NULL OR
       `estado_cobertura` NOT IN
         ('COMPLETA', 'PARCIAL', 'DESCONOCIDA') OR
       `trazabilidad_verificada` IS NULL OR
       `cobertura_completa` IS NULL OR
       `cobertura_completa` IS DISTINCT FROM
         (`estado_cobertura` = 'COMPLETA') OR
       (`registros_elegibles` IS NOT NULL AND
         (`registros_elegibles` < 0 OR
          `registros_elegibles` != `cantidad_sab_diaria`)) OR
       `registros_cuarentena` < 0 OR
       `archivos_excluidos` < 0 OR
       (`estado_cobertura` IN ('COMPLETA', 'PARCIAL')
         AND (
           `trazabilidad_verificada` IS NOT TRUE OR
           NULLIF(TRIM(`version_silver`), '') IS NULL OR
           NULLIF(TRIM(`scope_quality_id`), '') IS NULL OR
           `registros_elegibles` IS NULL OR
           `registros_cuarentena` IS NULL OR
           `archivos_excluidos` IS NULL
         )) OR
       (`estado_cobertura` = 'COMPLETA'
         AND (`registros_cuarentena` != 0 OR

       (`estado_cobertura` = 'PARCIAL'
         AND `registros_cuarentena` = 0
         AND `archivos_excluidos` = 0)

-- NO ejecutar en GCP sin validación y autorización de Quality.
--
-- Parámetros requeridos:
-- @product, @batch_id, @run_id, @source_version, @payload_sha256
--
-- IMPORTANTE:
-- Este preflight no reserva el batch ni evita carreras concurrentes.
-- Debe integrarse en un mecanismo transaccional autorizado.

ASSERT (
  SELECT COUNT(*) = 0
  FROM `gold_analytics_control.incremental_batches`
  WHERE product = @product
    AND batch_id = @batch_id
    AND (
      run_id != @run_id
      OR source_version != @source_version
      OR payload_sha256 != @payload_sha256
    )
) AS 'ERROR: batch_id reutilizado con trazabilidad diferente';

  SELECT COUNT(*) <= 1
  FROM `gold_analytics_control.incremental_batches`
  WHERE product = @product
    AND batch_id = @batch_id
) AS 'ERROR: registros de batch duplicados';

ASSERT (
  SELECT COUNT(*) = 0
  FROM `gold_analytics_control.incremental_batches`
  WHERE product = @product
    AND batch_id = @batch_id
    AND status NOT IN ('COMPLETED')
) AS 'ERROR: batch existente sin finalizacion confirmada';

-- Un batch COMPLETED con la misma identidad debe tratarse como REPLAY.
-- El orquestador debe omitir el MERGE para ese caso.
-- No basta con ejecutar este preflight para garantizarlo.
-- La atomicidad, exclusión concurrente y auditoría siguen pendientes.

"""Simulador transaccional Gold -> BigQuery. Sin acceso a GCP."""

import hashlib
import json
import sqlite3

CLAVES = {
    "mercado_valor_diario": ("nemonico", "fecha"),
    "actividad_sab_diaria": ("fecha", "codigo_sab"),
    "empresa_trimestre": ("id_registro",),
}


def aplicar_batch(conexion, producto, batch_id, run_id, version, filas):
    if producto not in CLAVES:
        raise ValueError("Producto Gold no permitido")
    if not all(isinstance(x, str) and x.strip()
               for x in (batch_id, run_id, version)):
        raise ValueError("Trazabilidad incompleta")
    if not isinstance(filas, list) or not filas:
        raise ValueError("Batch vacío")

    claves = CLAVES[producto]
    vistos = set()
    for fila in filas:
        if not isinstance(fila, dict):
            raise ValueError("Fila inválida")
        if fila.get("operation") != "UPSERT":
            raise ValueError("DELETE bloqueado")
        valores = tuple(fila.get(c) for c in claves)
        if any(v is None or v == "" for v in valores):
            raise ValueError("Clave nula")
        identidad = json.dumps(valores, ensure_ascii=False)
        if identidad in vistos:
            raise ValueError("Clave duplicada")
        vistos.add(identidad)

    contenido = json.dumps(
        filas, sort_keys=True, ensure_ascii=False, default=str
    )
    huella = hashlib.sha256(contenido.encode()).hexdigest()

    with conexion:
        conexion.execute("""
            CREATE TABLE IF NOT EXISTS batches (
                product TEXT, batch_id TEXT, run_id TEXT,
                version TEXT, sha256 TEXT,
                PRIMARY KEY (product, batch_id)
            )
        """)
        conexion.execute("""
            CREATE TABLE IF NOT EXISTS datos (
                product TEXT, clave TEXT, contenido TEXT,
                PRIMARY KEY (product, clave)
            )
        """)

        anterior = conexion.execute(
            "SELECT run_id, version, sha256 FROM batches "
            "WHERE product=? AND batch_id=?",
            (producto, batch_id),
        ).fetchone()

        if anterior:
            if anterior != (run_id, version, huella):
                raise ValueError("Batch ID reutilizado con otro contenido")
            return {"status": "REPLAY", "changes": 0}

        cambios = 0
        for fila in filas:
            identidad = json.dumps(
                [fila[c] for c in claves], ensure_ascii=False
            )
            datos = json.dumps(fila, sort_keys=True, default=str)
            actual = conexion.execute(
                "SELECT contenido FROM datos WHERE product=? AND clave=?",
                (producto, identidad),
            ).fetchone()

            if actual is None or actual[0] != datos:
                cambios += 1

            conexion.execute("""
                INSERT INTO datos (product, clave, contenido)
                VALUES (?, ?, ?)
                ON CONFLICT(product, clave)
                DO UPDATE SET contenido=excluded.contenido
            """, (producto, identidad, datos))

        conexion.execute(
            "INSERT INTO batches VALUES (?, ?, ?, ?, ?)",
            (producto, batch_id, run_id, version, huella),
        )

    return {"status": "COMPLETED", "changes": cambios}

    conexion, producto, batch_id, run_id,

"""Verificación SHA-256 local. No valida Parquet oficial."""

import hashlib
import json
import re


def calcular_sha256(filas):
    if not isinstance(filas, list) or not filas:
        raise ValueError("Payload vacío o inválido")

    contenido = json.dumps(
        filas,
        sort_keys=True,
        ensure_ascii=False,
        separators=(",", ":"),
        default=str,
    )
    return hashlib.sha256(contenido.encode("utf-8")).hexdigest()


def verificar_sha256(filas, hash_declarado):
    if not isinstance(hash_declarado, str):
        raise ValueError("SHA-256 inválido")

    if not re.fullmatch(r"[0-9a-fA-F]{64}", hash_declarado):
        raise ValueError("Formato SHA-256 inválido")

    calculado = calcular_sha256(filas)

    if calculado.lower() != hash_declarado.lower():
        raise ValueError("SHA-256 no coincide con el contenido")

    return True

class IncrementalExecutionError(Exception):
    pass


def exigir(condicion, mensaje):

    if not condicion:
        raise IncrementalExecutionError(mensaje)


def imprimir_seccion(titulo):

    print("\n" + "=" * 70)
    print(titulo)
    print("=" * 70)


def fecha_actual_utc():

    return datetime.now(
        timezone.utc
    ).isoformat()


def alinear_esquema(dataframe, referencia):

    exigir(
        dict(dataframe.dtypes)
        == dict(referencia.dtypes),
        "Esquemas incompatibles para comparacion.",
    )

    return dataframe.select(
        *referencia.columns
    )


def comprobar_equivalencia(
    izquierda,
    derecha,
    nombre,
):

    izquierda = alinear_esquema(
        izquierda,
        derecha,
    )

    solo_izquierda = (
        izquierda.exceptAll(derecha).count()
    )

    solo_derecha = (
        derecha.exceptAll(izquierda).count()
    )

    print(f"\nComparacion: {nombre}")

    print(
        "Solo izquierda:",
        solo_izquierda,
    )

    print(
        "Solo derecha:",
        solo_derecha,
    )

    exigir(
        solo_izquierda == 0
        and solo_derecha == 0,
        f"Equivalencia rechazada: {nombre}",
    )

    return {

        "comparacion": nombre,

        "solo_izquierda": solo_izquierda,

        "solo_derecha": solo_derecha,

        "equivalencia_aprobada": True,

    }

def cargar_alcance(
    escenario,
    registro,
    huella_registro,
    inventario_gold,
    inventario_silver,
):

    ruta = (

        RAIZ_REPORTES

        / f"alcance_gold_bcrp_{escenario}.json"

    )

    alcance, huella = cargar_json(ruta)

    evidencia = cargar_evidencia(

        escenario,

        registro,

        huella_registro,

    )

    exigir(
        alcance["entorno"] == "PRUEBA",
        "Entorno incompatible.",
    )

    exigir(
        alcance["escenario"] == escenario,
        "Escenario incompatible.",
    )

    exigir(
        alcance["run_id"]
        == evidencia["reporte"]["run_id"],
        "run_id incompatible.",
    )

    exigir(
        alcance["estado"] in {
            "GOLD_SIN_REGISTROS_AFECTADOS",
            "GOLD_ALCANCE_CANDIDATO_RESUELTO",
        },
        "Estado de alcance incompatible.",
    )

    exigir(
        alcance["alcance_candidato_resuelto"]
        is True,
        "Alcance candidato pendiente.",
    )

    exigir(
        alcance["validacion_operativa_aprobada"]
        is True,
        "Validacion operativa pendiente.",
    )

    exigir(
        alcance["huella_registro_sha256"]
        == huella_registro,
        "El registro fue modificado.",
    )

    exigir(
        alcance["huella_evidencia_operativa_sha256"]
        == evidencia["huella_reporte_sha256"],
        "La evidencia operativa fue modificada.",
    )

    exigir(
        alcance["huella_manifest_sha256"]
        == evidencia["reporte"]["huella_manifest_sha256"],
        "Manifest incompatible.",
    )

    exigir(
        alcance["inventario_gold_base"]
        == inventario_gold,
        "Gold base incompatible con el alcance.",
    )

    exigir(
        alcance["fuentes_silver_fijadas"]
        == inventario_silver,
        "Fuentes Silver incompatibles.",
    )

    exigir(
        alcance["version_silver_anterior"]
        == evidencia["version_anterior"]["version"],
        "Version Silver anterior incompatible.",
    )

    exigir(
        alcance["version_silver_publicada"]
        == evidencia["version_publicada"]["version"],
        "Version Silver publicada incompatible.",
    )

    exigir(
        alcance["ruta_silver_publicada"]
        == evidencia["version_publicada"]["ruta_silver"],
        "Ruta Silver incompatible.",
    )

    exigir(
        alcance["detalle_cambios_silver"]
        == evidencia["dataset"]["detalle_cambios"],
        "Detalle de cambios incompatible.",
    )

    exigir(
        alcance["publicacion_productiva"] is False,
        "Publicacion productiva bloqueada.",
    )

    exigir(
        alcance["autorizacion_publicacion_gold"]
        is False,
        "No se permite publicar Gold.",
    )

    exigir(

        is False,

    )

    # Esta primera version del ejecutor soporta
    # mercado afectado e impacto sin candidatos.
    # No ignora dependencias indirectas.

    exigir(
        alcance["cantidad_claves_impacto"] == 0,

        "Este escenario requiere ampliar el ejecutor.",
    )

    exigir(
        set(alcance["productos_con_recalculo"])
        <= {"mercado_valor_diario"},
        "Existe un producto no soportado.",
    )

    return alcance, huella, evidencia

def construir_claves_mercado(
    spark,
    alcance,
):

    tareas = [

        tarea

        for tarea in alcance["tareas"]

        if tarea["producto_gold"]
        == "mercado_valor_diario"

    ]

    exigir(
        len(tareas) == 1,
        "Se esperaba una tarea de mercado.",
    )

    tarea = tareas[0]

    claves = tarea["claves_gold_afectadas"]

    exigir(
        len(claves)
        == alcance["cantidad_claves_mercado"],
        "Cantidad de claves incompatible.",
    )

    filas = [

        (

            item["nemonico"],

            date.fromisoformat(item["fecha"]),

        )

        for item in claves

    ]

    esquema = StructType([

        StructField(
            "nemonico",
            StringType(),
            False,
        ),

        StructField(
            "fecha",
            DateType(),
            False,
        ),

    ])

    dataframe = spark.createDataFrame(

        filas,

        schema=esquema,

    )

    validar_clave(

        dataframe,

        CLAVE_MERCADO,

        "claves candidatas mercado",

    )

    return dataframe, tarea

def validar_alcance_mercado(
    claves,
    cotizaciones,
    mercado_base,
    alcance,
):

    fechas = [

        date.fromisoformat(valor)

        for valor in alcance[
            "fechas_silver_afectadas"
        ]

    ]

    claves_silver = (

        cotizaciones.filter(

            F.col("fecha_cotizacion").isin(fechas)

        )

        .select(

            F.col("valor").alias("nemonico"),

            F.col("fecha_cotizacion").alias("fecha"),

        )

    )

    claves_gold = (

        mercado_base.filter(

            F.col("fecha").isin(fechas)

        )

        .select(*CLAVE_MERCADO)

    )

    comprobar_equivalencia(

        claves,

        claves_silver,

        "Claves alcance vs cotizaciones Silver",

    )

    comprobar_equivalencia(

        claves,

        claves_gold,

        "Claves alcance vs mercado historico",

    )

# 7. RECALCULO INCREMENTAL DE MERCADO
# ============================================================

def recalcular_mercado(
    cotizaciones,
    tc_nuevo,
    embig,
    puente,
    mercado_base,
    claves,
):

    cotizaciones_afectadas = (

        cotizaciones.alias("c")

        .join(

            claves.alias("k"),

            (

                F.col("c.valor")
                == F.col("k.nemonico")

            )

            & (

                F.col("c.fecha_cotizacion")
                == F.col("k.fecha")

            ),

            "left_semi",

        )

    )

    # Reutilizacion del constructor historico.
    # Solo recibe las cotizaciones afectadas.

    recalculados = construir_mercado(

        cotizaciones_afectadas,

        tc_nuevo,

        embig,

        puente,

    )

    recalculados = alinear_esquema(

        recalculados,

        mercado_base,

    ).persist()

    validar_clave(

        recalculados,

        CLAVE_MERCADO,

        "mercado recalculado",

    )

    comprobar_equivalencia(

        recalculados.select(*CLAVE_MERCADO),

        claves,

        "Claves recalculadas vs alcance",

    )

    anteriores = mercado_base.join(

        claves,

        CLAVE_MERCADO,

        "left_semi",

    )

    columnas_no_modificables = [

        columna

        for columna in mercado_base.columns

        if columna not in COLUMNAS_MACRO_MODIFICABLES

    ]

    comprobar_equivalencia(

        recalculados.select(
            *columnas_no_modificables
        ),

        anteriores.select(
            *columnas_no_modificables
        ),

        "Columnas ajenas al tipo de cambio",

    )

    # Registros Gold que no necesitan recalculo.

    conservados = mercado_base.join(

        claves,

        CLAVE_MERCADO,

        "left_anti",

    )

    candidato = conservados.unionByName(

        recalculados

    ).persist()

    diferencias = recalculados.exceptAll(

        anteriores.select(*recalculados.columns)

    ).count()

    return (

        candidato,

        recalculados,

        diferencias,

    )

# 8. ESCRITURA DE VERSION GOLD CANDIDATA
# ============================================================

def escribir_candidata(
    escenario,
    candidato,
    particiones,
):

    RAIZ_CANDIDATAS.mkdir(

        parents=True,

        exist_ok=True,

    )

    exigir(
        not RAIZ_CANDIDATAS.is_symlink(),
        "Raiz candidata simbolica no permitida.",
    )

    ruta_candidata = (

        RAIZ_CANDIDATAS

        / f"gold_bcrp_{escenario}_v1"

    )

    exigir(
        not ruta_candidata.exists()
        and not ruta_candidata.is_symlink(),
        f"La version candidata ya existe: {ruta_candidata}",
    )

    origen_mercado = (

        GOLD_HISTORICO

        / PRODUCTOS_GOLD["mercado_valor_diario"]

    )

    rutas_excluidas = {

        origen_mercado

        / f"anio={particion['anio']}"

        / f"mes={particion['mes']}"

        for particion in particiones

    }

    exigir(
        all(
            ruta.is_dir()
            for ruta in rutas_excluidas
        ),
        "Una particion candidata no existe en Gold base.",
    )

    def ignorar_particiones(directorio, nombres):

        directorio = Path(directorio)

        return [

            nombre

            for nombre in nombres

            if directorio / nombre
            in rutas_excluidas

        ]

    # Copiamos Gold historico, excluyendo solamente
    # las particiones de mercado que deben escribirse.

    shutil.copytree(

        GOLD_HISTORICO,

        ruta_candidata,

        ignore=ignorar_particiones,

    )

    destino_mercado = (

        ruta_candidata

        / PRODUCTOS_GOLD["mercado_valor_diario"]

    )

    particiones_escritas = []

    for particion in particiones:

        anio = particion["anio"]

        mes = particion["mes"]

        destino_particion = (

            destino_mercado

            / f"anio={anio}"

            / f"mes={mes}"

        )

        exigir(
            not destino_particion.exists(),
            "La particion destino ya existe.",
        )

        datos_particion = candidato.filter(

            (F.col("anio") == anio)

            & (F.col("mes") == mes)

        )

        cantidad = datos_particion.count()

        exigir(
            cantidad > 0,
            "La particion candidata esta vacia.",
        )

        (

            datos_particion

            .drop("anio", "mes")

            .coalesce(1)

            .write

            .mode("errorifexists")

            .option("compression", "snappy")

            .parquet(str(destino_particion))
        )

        particiones_escritas.append({

            "anio": anio,

            "mes": mes,

            "registros_particion": cantidad,

        })

        print(

            f"Particion escrita: {anio}/{mes} "

            f"-> {cantidad} registros"

        )

    return ruta_candidata, particiones_escritas

# 9. VERIFICAR PRODUCTOS Y PARTICIONES CONSERVADAS
# ============================================================

def verificar_productos_conservados(
    ruta_candidata,
    inventario_gold,
):

    conservados = []

    for producto, relativa in PRODUCTOS_GOLD.items():

        if producto == "mercado_valor_diario":
            continue

        actual = inventariar_dataset(

            producto,

            ruta_candidata / relativa,

        )

        esperado = inventario_gold[producto]

        exigir(
            actual["huella_dataset_sha256"]
            == esperado["huella_dataset_sha256"],
            f"Producto no afectado modificado: {producto}",
        )

        conservados.append(producto)

    return conservados


def verificar_particiones_conservadas(
    ruta_candidata,
    particiones,
):

    from version_registry import construir_integridad

    origen = (

        GOLD_HISTORICO

        / PRODUCTOS_GOLD["mercado_valor_diario"]

    )

    destino = (

        ruta_candidata

        / PRODUCTOS_GOLD["mercado_valor_diario"]

    )

    afectadas = {

        f"anio={item['anio']}/mes={item['mes']}"

        for item in particiones

    }

    verificadas = 0

    for ruta in sorted(

        origen.glob("anio=*/mes=*")

    ):

        relativa = ruta.relative_to(origen)

        if relativa.as_posix() in afectadas:
            continue

        exigir(
            construir_integridad(ruta)

            == construir_integridad(
                destino / relativa
            ),

            f"Particion no afectada modificada: {relativa}",
        )

        verificadas += 1

    return verificadas

def ejecutar_escenario(
    spark,
    escenario,
):

    imprimir_seccion(

        "EJECUCION BCRP: " + escenario.upper()

    )

    registro, huella_registro = cargar_registro(

        RUTA_REGISTRO

    )

    inventario_gold = inventariar_gold()

    inventario_silver = inventariar_fuentes_silver()

    alcance, huella_alcance, evidencia = (

        cargar_alcance(

            escenario,

            registro,

            huella_registro,

            inventario_gold,

            inventario_silver,

        )

    )

    ruta_reporte = (

        RAIZ_REPORTES

        / f"ejecucion_gold_bcrp_{escenario}.json"

    )

    # Control de idempotencia de ejecuciones locales.

    exigir(
        not ruta_reporte.is_symlink(),
        "No se permite un reporte simbolico.",
    )

    if ruta_reporte.exists():

        imprimir_seccion(
            "CONTROL DE IDEMPOTENCIA"
        )

        previo, huella_previo = cargar_json(
            ruta_reporte
        )

        controles = {
            "entorno": "PRUEBA",
            "escenario": escenario,
            "run_id": alcance["run_id"],
            "huella_manifest_sha256": (
                alcance["huella_manifest_sha256"]
            ),
            "huella_registro_sha256": huella_registro,
            "huella_alcance_sha256": huella_alcance,
            "version_silver_evaluada": (
                alcance["version_silver_publicada"]
            ),
            "version_gold_base": (
                alcance["version_gold_base"]
            ),
            "inventario_gold_base": inventario_gold,
        }

        for campo, esperado in controles.items():

            exigir(
                previo.get(campo) == esperado,
                f"Conflicto de idempotencia: {campo}",
            )

        exigir(
            previo.get("estado") in {
                "GOLD_SIN_CAMBIOS",
                "GOLD_CANDIDATO_PRUEBA_VALIDADO",
            },
            "La ejecucion anterior no fue exitosa.",
        )

        for campo in [
            "validacion_operativa_aprobada",
            "alcance_efectivo_resuelto",
            "equivalencia_aprobada",
            "gold_historico_conservado",
            "fuentes_silver_conservadas",
            "ejecucion_experimental_completada",
        ]:

            exigir(
                previo.get(campo) is True,
                f"Control anterior no aprobado: {campo}",
            )

        exigir(
            previo.get("publicacion_productiva")
            is False,
            "Publicacion productiva no permitida.",
        )

        exigir(
            previo.get("autorizacion_publicacion_gold")
            is False,
            "Autorizacion de publicacion incompatible.",
        )

        comparaciones = previo.get("comparaciones")

        exigir(
            isinstance(comparaciones, list)
            and len(comparaciones) >= 2,
            "Comparaciones anteriores incompletas.",
        )

        for comparacion in comparaciones:

            exigir(
                comparacion.get("equivalencia_aprobada")
                is True
                and comparacion.get("solo_izquierda") == 0
                and comparacion.get("solo_derecha") == 0,
                "Equivalencia anterior no aprobada.",
            )

        ruta_esperada = (
            RAIZ_CANDIDATAS
            / f"gold_bcrp_{escenario}_v1"
        )

        if previo["estado"] == (
            "GOLD_CANDIDATO_PRUEBA_VALIDADO"
        ):

            exigir(
                previo.get("ruta_gold_candidata")
                == ruta_esperada.as_uri(),
                "Ruta candidata incompatible.",
            )

            exigir(
                previo.get("version_gold_candidata")
                == ruta_esperada.name,
                "Version candidata incompatible.",
            )

            exigir(
                ruta_esperada.is_dir()
                and not ruta_esperada.is_symlink(),
                "La candidata anterior no esta disponible.",
            )

            inventario_actual = {

                producto: inventariar_dataset(
                    producto,
                    ruta_esperada / relativa,
                )

                for producto, relativa
                in PRODUCTOS_GOLD.items()

            }

            exigir(
                inventario_actual
                == previo.get("inventario_gold_candidata"),
                "La candidata anterior fue modificada.",
            )

            print(
                "OK: integridad Gold candidata verificada."
            )

        else:

            exigir(
                previo.get("version_gold_candidata")
                is None
                and previo.get("ruta_gold_candidata")
                is None
                and previo.get("inventario_gold_candidata")
                is None,
                "El escenario sin cambios tiene una candidata.",
            )

            exigir(
                previo.get("registros_mercado_modificados")
                == 0,
                "Resultado anterior incompatible.",
            )

            exigir(
                not ruta_esperada.exists()
                and not ruta_esperada.is_symlink(),
                "Existe una candidata inesperada.",
            )

        exigir(
            inventariar_gold() == inventario_gold,
            "Gold historico fue modificado.",
        )

        exigir(
            inventariar_fuentes_silver()
            == inventario_silver,
            "Las fuentes Silver fueron modificadas.",
        )

        verificar_integridad_version(
            evidencia["version_anterior"]
        )

        verificar_integridad_version(

        )

        _, huella_final = cargar_json(
            ruta_reporte
        )

        exigir(
            huella_final == huella_previo,
            "El reporte anterior fue modificado.",
        )

        resultado = dict(previo)

        resultado["estado_anterior"] = previo["estado"]

        resultado["estado"] = (
            "GOLD_OMITIDO_IDEMPOTENCIA"
        )

        resultado["registros_mercado_recalculados"] = 0

        resultado["registros_mercado_modificados"] = 0

        resultado["particiones_escritas"] = []

        resultado["ejecucion_omitida"] = True

        resultado["huella_reporte_anterior_sha256"] = (
            huella_previo
        )

        print("OK: run_id coincidente.")
        print("OK: huella manifest coincidente.")
        print("OK: ejecucion anterior validada.")
        print("OK: integridad de fuentes conservada.")
        print("OK: reporte anterior conservado.")
        print("Registros recalculados: 0")
        print("Registros modificados: 0")
        print("ESTADO: GOLD_OMITIDO_IDEMPOTENCIA")

        return resultado

    if alcance["cantidad_claves_mercado"] > 0:

        posible_candidata = (

            RAIZ_CANDIDATAS

            / f"gold_bcrp_{escenario}_v1"

        )

        exigir(
            not posible_candidata.exists()
            and not posible_candidata.is_symlink(),

            "Inspeccionar antes de continuar.",
        )

    dataframes = []

    comparaciones = []

    ruta_candidata = None

    particiones_escritas = []

    particiones_conservadas = 0

    productos_conservados = []

    try:

        cotizaciones = spark.read.parquet(

            inventario_silver["cotizaciones"]["ruta"]

        ).persist()

        dataframes.append(cotizaciones)

        embig = spark.read.parquet(

            inventario_silver["embig"]["ruta"]

        ).persist()

        dataframes.append(embig)

        puente = spark.read.parquet(

            inventario_silver[
                "puente_ruc_nemonico"
            ]["ruta"]

        ).persist()

        dataframes.append(puente)

        tc_nuevo = leer_version(

        mercado_base = spark.read.parquet(

            ]["ruta"]

def validar_configuracion_planificador():

    errores = []

    asegurar_registro_valido()

    if set(ESTRATEGIAS_GOLD) != set(PRODUCTOS_GOLD):

        errores.append(
            "Las estrategias Gold no coinciden "
            "con los productos registrados."
        )

    for producto, estrategia in (
        ESTRATEGIAS_GOLD.items()
    ):

        if producto not in PRODUCTOS_GOLD:

            continue

        campos_requeridos = {

            "estrategia",

            "unidad_recalculo",

            "fuentes_silver_requeridas",

            "fuentes_gold_requeridas",

            "acciones",

            "controles_especificos",

        }

        faltantes = campos_requeridos - set(
            estrategia
        )

        if faltantes:

            errores.append(
                f"{producto}: campos faltantes "
                f"{sorted(faltantes)}."
            )

            continue

        fuentes_silver = estrategia[
            "fuentes_silver_requeridas"
        ]

        fuentes_gold = estrategia[
            "fuentes_gold_requeridas"
        ]

        fuentes_silver_esperadas = {

            dataset

            for dataset, dependencia
            in DEPENDENCIAS_SILVER_GOLD.items()

            if producto in dependencia[
                "productos"
            ]

        }

        fuentes_gold_esperadas = {

            origen

            for origen, destinos
            in DEPENDENCIAS_GOLD_GOLD.items()

            if producto in destinos

        }

        if set(fuentes_silver) != (
            fuentes_silver_esperadas
        ):

            errores.append(
                f"{producto}: fuentes Silver "
                "inconsistentes."
            )

        if set(fuentes_gold) != (
            fuentes_gold_esperadas
        ):

            errores.append(
                f"{producto}: fuentes Gold "
                "inconsistentes."
            )

        if len(fuentes_silver) != len(
            set(fuentes_silver)
        ):

            errores.append(
                f"{producto}: fuentes Silver "
                "duplicadas."
            )

        if len(fuentes_gold) != len(
            set(fuentes_gold)
        ):

            errores.append(
                f"{producto}: fuentes Gold "
                "duplicadas."
            )

        if not estrategia["acciones"]:

            errores.append(
                f"{producto}: acciones vacias."
            )

        if not estrategia["unidad_recalculo"]:

            errores.append(
                f"{producto}: unidad de "
                "recalculo vacia."
            )

        if not estrategia["controles_especificos"]:

            errores.append(
                f"{producto}: controles vacios."
            )

    regla_impacto = REGLAS_PROPAGACION.get(
        (
            "mercado_valor_diario",
            "impacto_hechos_valor",
        ),
        {},
    )

    estrategia_impacto = ESTRATEGIAS_GOLD.get(
        "impacto_hechos_valor",
        {},
    )

    if estrategia_impacto.get(
        "ventana_dias"
    ) != regla_impacto.get(
        "ventana_dias"
    ):

        errores.append(
            "La ventana de impacto no coincide "
            "con el registro de dependencias."
        )

    if estrategia_impacto.get(
        "excluir_dia_evento"
    ) != regla_impacto.get(
        "excluir_dia_evento"
    ):

        errores.append(
            "La exclusion del dia del evento "
            "no coincide con el registro."
        )

    codigos = [

        requisito["codigo"]

        for requisito in REQUISITOS_OPERATIVOS

    ]

    if len(codigos) != len(set(codigos)):

        errores.append(
            "Existen requisitos operativos "
            "duplicados."
        )

    return errores


def asegurar_configuracion_valida():

    errores = validar_configuracion_planificador()

    if errores:

        raise PlannerError(
            "Configuracion invalida: "
            + "; ".join(errores)
        )

def obtener_predecesores_gold(producto):

    if producto not in PRODUCTOS_GOLD:

        raise PlannerError(
            f"Producto Gold desconocido: "
            f"{producto}"
        )

    predecesores = {

        origen

        for origen, destinos
        in DEPENDENCIAS_GOLD_GOLD.items()

        if producto in destinos

    }

    return [

        nombre

        for nombre in ORDEN_EJECUCION_GOLD

        if nombre in predecesores

    ]


def obtener_reglas_predecesores(
    producto,
    productos_planificados,
):

    predecesores = obtener_predecesores_gold(
        producto
    )

    reglas = []

    for origen in predecesores:

        if origen not in productos_planificados:

            continue

        regla = copy.deepcopy(
            REGLAS_PROPAGACION[
                (origen, producto)
            ]
        )

        reglas.append(
            {
                "producto_origen": origen,

                "producto_destino": producto,

                **regla,

                "requiere_cambios_efectivos": True,

            }
        )

    return reglas

def validar_orden_tareas(tareas):

    errores = []

    productos = [

        tarea["producto_gold"]

        for tarea in tareas

    ]

    if len(productos) != len(set(productos)):

        errores.append(
            "El plan contiene productos duplicados."
        )

    posiciones = {

        producto: indice

        for indice, producto
        in enumerate(productos)

    }

    orden_esperado = [

        producto

        for producto in ORDEN_EJECUCION_GOLD

        if producto in posiciones

    ]

    if productos != orden_esperado:

        errores.append(
            "El orden de tareas no coincide "
            "con el orden Gold registrado."
        )

    for tarea in tareas:

        producto = tarea["producto_gold"]

        for predecesor in tarea[
            "predecesores_planificados"
        ]:

            if predecesor not in posiciones:

                errores.append(
                    f"{producto}: predecesor "
                    f"ausente: {predecesor}."
                )

                continue

            if posiciones[predecesor] >= (
                posiciones[producto]
            ):

                errores.append(
                    f"{producto}: orden incompatible "
                    f"con {predecesor}."
                )

    return errores

        )

            "Existen detalles externos de cambios "

        )

        ESTADO_PLANIFICADO

def main():

    parser = argparse.ArgumentParser(

        description=(
            "Planificador incremental "
            "preliminar Silver-Gold."
        ),

    )

    grupo = (
        parser.add_mutually_exclusive_group(
            required=True
        )
    )

    grupo.add_argument(

        "--manifest",

        type=Path,

        help=(
            "Ruta del manifest Silver."
        ),

    )

    grupo.add_argument(

        "--mostrar-configuracion",

        action="store_true",

        help=(
            "Mostrar estrategias y requisitos "
            "del planificador."
        ),

    )

    parser.add_argument(

        "--entorno",

        choices=[
            "PRUEBA",
            "PRODUCCION",
        ],

        default="PRUEBA",

    )

    parser.add_argument(

        "--resumen",

        action="store_true",

        help=(
            "Mostrar una version compacta "
            "del plan preliminar."
        ),

    )

    args = parser.parse_args()

    if args.mostrar_configuracion:

        try:

            resultado = obtener_configuracion()

        except (
            PlannerError,
            DependencyError,
        ) as error:

            resultado = {
                "estado": ESTADO_RECHAZADO,

                "autorizacion_ejecucion_gold": (
                    False
                ),

                "autorizacion_publicacion_gold": (
                    False
                ),

                "errores": [
                    str(error)
                ],
            }

    else:

        resultado = planificar_archivo_manifest(

            ruta=args.manifest,

            entorno_esperado=args.entorno,

        )

        if args.resumen:

            resultado = obtener_resumen_plan(
                resultado
            )

    print(
        json.dumps(

            resultado,

            indent=4,

            ensure_ascii=False,

            allow_nan=False,

        )
    )

    if resultado["estado"] == ESTADO_RECHAZADO:

        return 1

    return 0


if __name__ == "__main__":

    sys.exit(main())

class ManifestError(Exception):
    """Error de lectura o validacion del manifest."""

def texto_valido(valor):
    return isinstance(valor, str) and bool(valor.strip())


def entero_no_negativo(valor):
    return type(valor) is int and valor >= 0


def fecha_iso_valida(valor):
    if not isinstance(valor, str):
        return False

    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", valor):
        return False

    try:
        date.fromisoformat(valor)
        return True
    except ValueError:
        return False


def timestamp_valido(valor):
    if not isinstance(valor, str):
        return False

    try:
        fecha = datetime.fromisoformat(
            valor.replace("Z", "+00:00")
        )

        return (
            "T" in valor
            and fecha.tzinfo is not None
            and fecha.utcoffset() is not None
        )

    except ValueError:
        return False


def identificador_valido(valor):
    if not texto_valido(valor):
        return False

    return bool(
        re.fullmatch(
            r"[A-Za-z0-9][A-Za-z0-9._-]{0,127}",
            valor,
        )
    )


def json_sin_duplicados(pares):
    resultado = {}

    for clave, valor in pares:

        if clave in resultado:
            raise ManifestError(
                f"Clave JSON duplicada: {clave}"
            )

        resultado[clave] = valor

    return resultado

def rechazar_constante_json(valor):
    raise ManifestError(
        f"Constante JSON no permitida: {valor}"
    )


def parsear_float_finito(valor):
    """
    Convierte un numero JSON a float y rechaza
    valores que excedan el rango finito de Python.
    """

    try:
        numero = float(valor)

    except (ValueError, OverflowError) as exc:
        raise ManifestError(
            f"Numero JSON invalido: {valor}"
        ) from exc

    if not math.isfinite(numero):
        raise ManifestError(
            f"Numero JSON no finito: {valor}"
        )

    return numero

def comprobar_campos(
    objeto,
    campos_obligatorios,
    contexto,
    errores,
):
    faltantes = campos_obligatorios - set(objeto)

    adicionales = set(objeto) - campos_obligatorios

    if faltantes:
        errores.append(
            f"{contexto}: campos faltantes: "
            f"{', '.join(sorted(faltantes))}"
        )

    if adicionales:
        errores.append(
            f"{contexto}: campos no reconocidos: "
            f"{', '.join(sorted(adicionales))}"
        )

def clave_canonica(clave):
    return json.dumps(
        clave,
        sort_keys=True,
        ensure_ascii=False,
        separators=(",", ":"),
        allow_nan=False,
    )


def validar_compatibilidad_json(manifest):
    """
    Verifica que un objeto Python sea compatible
    con una estructura JSON valida.

    Rechaza:
    - Numeros no finitos.
    - Objetos no serializables.
    - Referencias circulares.
    - Anidamiento excesivo.
    - Claves de diccionario que no sean cadenas.
    - Caracteres Unicode invalidos.

    No modifica el objeto recibido.
    """

    profundidad_maxima = 64
    objetos_activos = set()

    def inspeccionar(valor, profundidad=0):

        if profundidad > profundidad_maxima:
            raise ManifestError(
                "Anidamiento JSON excesivo."
            )

        if valor is None:
            return

        if type(valor) in (bool, int):
            return

        if type(valor) is float:

            if not math.isfinite(valor):
                raise ManifestError(
                    "Numero JSON no finito."
                )

            return

        if type(valor) is str:

            try:
                valor.encode("utf-8")

            except UnicodeEncodeError as exc:
                raise ManifestError(
                    "Cadena con caracteres Unicode invalidos."
                ) from exc

            return

        if type(valor) not in (dict, list):

            raise ManifestError(
                "Tipo Python no compatible con JSON: "
                f"{type(valor).__name__}."
            )

        identidad = id(valor)

        if identidad in objetos_activos:

            raise ManifestError(
                "Referencia circular en estructura JSON."
            )

        objetos_activos.add(identidad)

        try:

            if type(valor) is dict:

                for clave, contenido in valor.items():

                    if type(clave) is not str:

                        raise ManifestError(
                            "Las claves JSON deben ser cadenas."
                        )

                    inspeccionar(
                        clave,
                        profundidad + 1,
                    )

                    inspeccionar(
                        contenido,
                        profundidad + 1,
                    )

            else:

                for contenido in valor:

                    inspeccionar(
                        contenido,
                        profundidad + 1,
                    )

        finally:

            objetos_activos.remove(
                identidad
            )

    try:

        inspeccionar(manifest)

        json.dumps(
            manifest,
            ensure_ascii=False,
            allow_nan=False,
        )

    except ManifestError as exc:

        return [
            f"Estructura JSON invalida: {exc}"
        ]

    except (
        ValueError,
        TypeError,
        OverflowError,
        RecursionError,
    ) as exc:

        return [
            f"Estructura JSON invalida: {exc}"
        ]

    return []

# 6. LECTURA DEL MANIFEST Y HUELLA SHA-256
# ============================================================
def cargar_manifest(ruta, backend=None):

    try:
        if backend is not None:
            contenido = backend.leer_bytes(str(ruta))
        elif str(ruta).startswith("gs://"):
            config = cargar_configuracion()
            if config.backend != BACKEND_GCS:
                raise ManifestError(
                    "La lectura GCS requiere configuracion GCS."
                )
            contenido = crear_backend(config).leer_bytes(str(ruta))
        else:
            contenido = Path(ruta).read_bytes()

    except Exception as exc:
        raise ManifestError(
            f"No se pudo leer el manifest: {exc}"
        ) from exc

    huella = hashlib.sha256(contenido).hexdigest()

    try:
        manifest = json.loads(
            contenido.decode("utf-8"),
            object_pairs_hook=json_sin_duplicados,
            parse_constant=rechazar_constante_json,
            parse_float=parsear_float_finito,
        )

    except ManifestError:
        raise

    except RecursionError as exc:
        raise ManifestError(
            "JSON invalido: nivel de anidamiento excesivo."
        ) from exc

    except (
        UnicodeDecodeError,
        json.JSONDecodeError,
        ValueError,
        OverflowError,
    ) as exc:
        raise ManifestError(
            f"JSON invalido: {exc}"
        ) from exc

    if not isinstance(manifest, dict):
        raise ManifestError(
            "El manifest debe ser un objeto JSON."
        )

    return manifest, huella

# 7. VALIDACION ESTRUCTURAL DE RUTAS
# ============================================================

def validar_ruta(
    ruta,
    entorno,
    contexto,
    errores,
    sufijo_esperado=None,
    config=None,
):
    if not texto_valido(ruta):
        errores.append(
            f"{contexto}: ruta vacia o invalida."
        )
        return

    if ruta != ruta.strip():
        errores.append(
            f"{contexto}: espacios al inicio "
            "o final de la ruta no permitidos."
        )
        return

    if any(
        ord(caracter) < 32 or ord(caracter) == 127
        for caracter in ruta
    ):
        errores.append(
            f"{contexto}: caracteres de control "
            "no permitidos en la ruta."
        )
        return

    if "\\" in ruta:
        errores.append(
            f"{contexto}: barras invertidas "
            "no permitidas en la ruta."
        )
        return

    if "%" in ruta:
        errores.append(
            f"{contexto}: codificacion porcentual "
            "no permitida en la ruta."
        )
        return

    try:
        componentes = urlparse(ruta)

    except ValueError:
        errores.append(
            f"{contexto}: ruta malformada."
        )
        return

    esquema = componentes.scheme

    if entorno == "PRUEBA":

        if config is not None and config.backend == BACKEND_GCS:

            try:
                validar_uri_gcs(ruta)

                raices = (
                    config.raices_autorizadas_silver
                    if sufijo_esperado
                    else (config.control, config.manifests)
                )

                if not any(
                    uri_bajo_raiz(ruta, raiz, BACKEND_GCS)
                    for raiz in raices
                ):
                    errores.append(
                        f"{contexto}: ruta GCS no autorizada."
                    )

            except StorageConfigError as exc:
                errores.append(f"{contexto}: {exc}")

        elif esquema != "file":
            errores.append(
                f"{contexto}: PRUEBA solo acepta "
                "rutas locales file://."
            )

    elif entorno == "PRODUCCION":

        if esquema not in {"hdfs", "gs"}:
            errores.append(
                f"{contexto}: esquema de "
                "produccion no permitido."
            )

    if esquema == "file":

        if componentes.netloc not in {"", "localhost"}:
            errores.append(
                f"{contexto}: ruta file:// no local."
            )

    elif esquema in {"hdfs", "gs"}:

        if not componentes.netloc:
            errores.append(
                f"{contexto}: falta autoridad o bucket."
            )

    if not componentes.path.startswith("/"):
        errores.append(
            f"{contexto}: se requiere ruta absoluta."
        )

    if componentes.query or componentes.fragment:
        errores.append(
            f"{contexto}: query o fragment no permitido."
        )

    if any(
        segmento in {".", ".."}
        for segmento in componentes.path.split("/")
    ):
        errores.append(
            f"{contexto}: segmentos de ruta no permitidos."
        )

    if sufijo_esperado:

        ruta_normalizada = componentes.path.rstrip("/")

        if not ruta_normalizada.endswith(
            "/" + sufijo_esperado
        ):
            errores.append(
                f"{contexto}: ruta incompatible "
                f"con el catalogo: {sufijo_esperado}"
            )

# 8. VALIDACION DE PARTICIONES
# ============================================================

def validar_particiones(
    particiones,
    configuracion,
    total_cambios,
    contexto,
    errores,
):
    if not isinstance(particiones, list):
        errores.append(
            f"{contexto}: particiones_afectadas "
            "debe ser una lista."
        )
        return set()

    campos_esperados = configuracion["particiones"]

    if (
        campos_esperados
        and total_cambios > 0
        and not particiones
    ):
        errores.append(
            f"{contexto}: faltan particiones afectadas."
        )

    particiones_vistas = set()

    for indice, particion in enumerate(particiones):

        prefijo = (
            f"{contexto}.particiones_afectadas[{indice}]"
        )

        if not isinstance(particion, dict):
            errores.append(
                f"{prefijo}: debe ser un objeto."
            )
            continue

        if not particion:
            errores.append(
                f"{prefijo}: particion vacia."
            )
            continue

        if (
            campos_esperados is not None
            and set(particion) != set(campos_esperados)
        ):
            errores.append(
                f"{prefijo}: campos de particion "
                f"esperados: {campos_esperados}"
            )

        for campo, valor in particion.items():

            if not entero_no_negativo(valor):
                errores.append(
                    f"{prefijo}.{campo}: entero invalido."
                )
                continue

            if campo in {
                "anio",
                "ejercicio",
                "anio_presentacion",
            }:

                if not 1900 <= valor <= 9999:
                    errores.append(
                        f"{prefijo}.{campo}: anio invalido."
                    )

            elif campo == "mes":

                if not 1 <= valor <= 12:
                    errores.append(
                        f"{prefijo}: mes fuera de rango."
                    )

            elif campo in {
                "trimestre",
                "trimestre_num",
            }:

                if not 1 <= valor <= 4:
                    errores.append(
                        f"{prefijo}: trimestre invalido."
                    )

            elif campos_esperados is None:
                errores.append(
                    f"{prefijo}: particion no confirmada "
                    f"en el contrato: {campo}"
                )

        canonica = clave_canonica(particion)

        if canonica in particiones_vistas:
            errores.append(
                f"{prefijo}: particion duplicada."
            )

        particiones_vistas.add(canonica)

    return particiones_vistas

# 9. VALIDACION DE CLAVES DE NEGOCIO
# ============================================================

def validar_clave_operacion(
    clave,
    configuracion,
    contexto,
    errores,
):
    if not isinstance(clave, dict):
        errores.append(
            f"{contexto}: se esperaba objeto de clave."
        )
        return False

    campos_esperados = set(configuracion["clave"])

    if set(clave) != campos_esperados:
        errores.append(
            f"{contexto}: campos de clave incorrectos. "
            f"Esperados: {sorted(campos_esperados)}"
        )
        return False

    valido = True

    for campo, valor in clave.items():

        if campo in configuracion["fechas_clave"]:

            if not fecha_iso_valida(valor):
                errores.append(
                    f"{contexto}.{campo}: fecha invalida."
                )
                valido = False

        else:

            if not (
                texto_valido(valor)
                or type(valor) is int
            ):
                errores.append(
                    f"{contexto}.{campo}: valor "
                    "de clave invalido."
                )
                valido = False

    return valido


def validar_particion_de_clave(
    clave,
    configuracion,
    particiones_vistas,
    contexto,
    errores,
):
    if configuracion["particiones"] != ["anio", "mes"]:
        return

    for campo in configuracion["fechas_clave"]:

        valor = clave.get(campo)

        if not fecha_iso_valida(valor):
            continue

        fecha = date.fromisoformat(valor)

        particion = {
            "anio": fecha.year,
            "mes": fecha.month,
        }

        if clave_canonica(
            particion
        ) not in particiones_vistas:

            errores.append(
                f"{contexto}: la particion {particion} "
                "de la clave no fue declarada."
            )

# 10. VALIDACION DEL DETALLE DE CAMBIOS
# ============================================================

def validar_operaciones(
    operaciones,
    configuracion,
    particiones_vistas,
    contexto,
    errores,
):
    conteos = Counter()

    if not isinstance(operaciones, list):
        errores.append(
            f"{contexto}: claves_afectadas "
            "debe ser una lista."
        )
        return conteos

    claves_origen = set()
    claves_destino = set()

    operaciones_por_clave = {}

    for indice, cambio in enumerate(operaciones):

        prefijo = (
            f"{contexto}.claves_afectadas[{indice}]"
        )

        if not isinstance(cambio, dict):
            errores.append(
                f"{prefijo}: operacion invalida."
            )
            continue

        comprobar_campos(
            cambio,
            CAMPOS_OPERACION,
            prefijo,
            errores,
        )

        operacion = cambio.get("operacion")

        if (
            not isinstance(operacion, str)
            or operacion not in OPERACIONES_VALIDAS
        ):
            errores.append(
                f"{prefijo}: operacion no reconocida."
            )
            continue

        conteos[operacion] += 1

        anterior = cambio.get("clave_anterior")
        nueva = cambio.get("clave_nueva")

        anterior_valida = False
        nueva_valida = False

        if operacion == "INSERT":

            if anterior is not None:
                errores.append(
                    f"{prefijo}: INSERT requiere "
                    "clave_anterior null."
                )

            nueva_valida = validar_clave_operacion(
                nueva,
                configuracion,
                prefijo + ".clave_nueva",
                errores,
            )

        elif operacion == "UPDATE":

            anterior_valida = validar_clave_operacion(
                anterior,
                configuracion,
                prefijo + ".clave_anterior",
                errores,
            )

            nueva_valida = validar_clave_operacion(
                nueva,
                configuracion,
                prefijo + ".clave_nueva",
                errores,
            )

        elif operacion == "DELETE":

            anterior_valida = validar_clave_operacion(
                anterior,
                configuracion,
                prefijo + ".clave_anterior",
                errores,
            )

            if nueva is not None:
                errores.append(
                    f"{prefijo}: DELETE requiere "
                    "clave_nueva null."
                )

        claves_cambio = set()

        if anterior_valida:

            canonica = clave_canonica(anterior)

            if canonica in claves_origen:
                errores.append(
                    f"{prefijo}: clave de origen repetida."
                )

            claves_origen.add(canonica)
            claves_cambio.add(canonica)

            validar_particion_de_clave(
                anterior,
                configuracion,
                particiones_vistas,
                prefijo + ".clave_anterior",
                errores,
            )

        if nueva_valida:

            canonica = clave_canonica(nueva)

            if canonica in claves_destino:
                errores.append(
                    f"{prefijo}: clave de destino repetida."
                )

            claves_destino.add(canonica)
            claves_cambio.add(canonica)

            validar_particion_de_clave(
                nueva,
                configuracion,
                particiones_vistas,
                prefijo + ".clave_nueva",
                errores,
            )

        for canonica in claves_cambio:

            if canonica in operaciones_por_clave:
                errores.append(
                    f"{prefijo}: clave presente "
                    "en multiples operaciones."
                )

            else:
                operaciones_por_clave[canonica] = indice

    return conteos

# 11. VALIDACION DE UN DATASET
# ============================================================

def validar_dataset(
    dataset,
    indice,
    entorno,
    errores,
    advertencias,
    config=None,
):
    contexto = f"datasets[{indice}]"

    if not isinstance(dataset, dict):
        errores.append(
            f"{contexto}: debe ser un objeto."
        )
        return

    comprobar_campos(
        dataset,
        CAMPOS_DATASET,
        contexto,
        errores,
    )

    nombre = dataset.get("dataset")

    if not texto_valido(nombre):
        errores.append(
            f"{contexto}: identificador invalido."
        )
        return

    contexto = f"dataset '{nombre}'"

    if nombre not in CATALOGO_DATASETS:
        errores.append(
            f"{contexto}: dataset desconocido."
        )
        return

    configuracion = CATALOGO_DATASETS[nombre]

    validar_ruta(
        dataset.get("ruta_silver"),
        entorno,
        contexto + ".ruta_silver",
        errores,
        sufijo_esperado=configuracion["ruta"],
        config=config,
    )

    for campo in (
        "version_esquema",
        "version_anterior",
        "version_publicada",
    ):

        if not texto_valido(dataset.get(campo)):
            errores.append(
                f"{contexto}.{campo}: version invalida."
            )

    clave = dataset.get("clave")

    if clave != configuracion["clave"]:
        errores.append(
            f"{contexto}: clave incorrecta. "
            f"Esperada: {configuracion['clave']}"
        )

    conteos_validos = True

    for campo in CAMPOS_CONTEOS:

        if not entero_no_negativo(dataset.get(campo)):
            errores.append(
                f"{contexto}.{campo}: debe ser "
                "un entero no negativo."
            )
            conteos_validos = False

    if type(
        dataset.get("autorizacion_eliminaciones")
    ) is not bool:

        errores.append(
            f"{contexto}: autorizacion_eliminaciones "
            "debe ser booleana."
        )

    total_cambios = 0

    if conteos_validos:

        nuevos = dataset["registros_nuevos"]
        corregidos = dataset["registros_corregidos"]
        eliminados = dataset["registros_eliminados"]

        antes = dataset["registros_antes"]
        despues = dataset["registros_despues"]

        total_cambios = nuevos + corregidos + eliminados

        esperado = antes + nuevos - eliminados

        if despues != esperado:
            errores.append(
                f"{contexto}: conteo final inconsistente. "
                f"Esperado={esperado}, recibido={despues}."
            )

        if corregidos + eliminados > antes:
            errores.append(
                f"{contexto}: modificaciones exceden "
                "los registros anteriores."
            )

        if (
            eliminados > 0
            and dataset.get(
                "autorizacion_eliminaciones"
            ) is not True
        ):
            errores.append(
                f"{contexto}: eliminacion no autorizada."
            )

        if (
            total_cambios > 0
            and dataset.get("version_anterior")
            == dataset.get("version_publicada")
        ):
            errores.append(
                f"{contexto}: existen cambios sin "
                "una nueva version publicada."
            )

    particiones_vistas = validar_particiones(
        dataset.get("particiones_afectadas"),
        configuracion,
        total_cambios,
        contexto,
        errores,
    )

    operaciones = dataset.get("claves_afectadas")

    ruta_detalle = dataset.get("ruta_detalle_cambios")

    tiene_operaciones = operaciones is not None
    tiene_ruta = ruta_detalle is not None

    if tiene_operaciones and tiene_ruta:

        errores.append(
            f"{contexto}: se debe utilizar una sola "
            "modalidad de detalle de cambios."
        )

    if (
        total_cambios > 0
        and not tiene_operaciones
        and not tiene_ruta
    ):
        errores.append(
            f"{contexto}: falta el detalle de cambios."
        )

    if tiene_operaciones:

        conteos_operaciones = validar_operaciones(
            operaciones,
            configuracion,
            particiones_vistas,
            contexto,
            errores,
        )

        if conteos_validos:

            esperados = {
                "INSERT": nuevos,
                "UPDATE": corregidos,
                "DELETE": eliminados,
            }

            for operacion, esperado in esperados.items():

                recibido = conteos_operaciones[operacion]

                if recibido != esperado:
                    errores.append(
                        f"{contexto}: conteo {operacion} "
                        f"inconsistente. "
                        f"Esperado={esperado}, "
                        f"recibido={recibido}."
                    )

    if tiene_ruta:

        validar_ruta(
            ruta_detalle,
            entorno,
            contexto + ".ruta_detalle_cambios",
            errores,
            config=config,
        )

        advertencias.append(
            f"{contexto}: detalle externo pendiente "
            "de validacion operativa."
        )

    if not configuracion["consumido_gold"]:

        advertencias.append(
            f"{contexto}: dataset reconocido, "
            "pero no consumido por Gold actualmente."
        )

    if nombre == "principales_cuentas":

        advertencias.append(
            f"{contexto}: estabilidad de id_registro "
            "pendiente de confirmacion con Silver."
        )

    if configuracion["particiones"] is None:

        advertencias.append(
            f"{contexto}: contrato de particiones "
            "pendiente de confirmacion operativa."
        )

# 12. VALIDACION GENERAL DEL MANIFEST
# ============================================================
def validar_manifest(
    manifest,
    entorno_esperado="PRUEBA",
    config=None,
):
    errores = []
    advertencias = []

    if not isinstance(manifest, dict):
        return (
            ["El manifest debe ser un objeto JSON."],
            [],
        )

    errores_json = validar_compatibilidad_json(
        manifest
    )

    if errores_json:

        return (
            errores_json,
            advertencias,
        )

    comprobar_campos(
        manifest,
        CAMPOS_MANIFEST,
        "manifest",
        errores,
    )

    version = manifest.get("version_manifest")

    if (
        not isinstance(version, str)
        or version not in VERSIONES_SOPORTADAS
    ):
        errores.append(
            "Version del manifest no soportada."
        )

    entorno = manifest.get("entorno")

    if (
        not isinstance(entorno, str)
        or entorno not in ENTORNOS_VALIDOS
    ):
        errores.append(
            "Entorno del manifest invalido."
        )

    elif entorno != entorno_esperado:

        errores.append(
            f"Entorno incompatible: "
            f"esperado={entorno_esperado}, "
            f"recibido={entorno}."
        )

    if entorno == "PRODUCCION":
        errores.append(
            "Produccion bloqueada: contrato provisional "
            "v0.1 pendiente de aprobacion."
        )

    if not identificador_valido(
        manifest.get("run_id")
    ):
        errores.append(
            "run_id invalido."
        )

    if manifest.get(
        "estado"
    ) != ESTADO_SILVER_APROBADO:

        errores.append(
            "Silver no se encuentra publicado y aprobado."
        )

    if manifest.get(
        "validacion_aprobada"
    ) is not True:

        errores.append(
            "validacion_aprobada debe ser true."
        )

    if not timestamp_valido(
        manifest.get("fecha_publicacion")
    ):
        errores.append(
            "fecha_publicacion invalida."
        )

    tipo_publicacion = manifest.get(
        "tipo_publicacion"
    )

    if (
        not isinstance(tipo_publicacion, str)
        or tipo_publicacion
        not in TIPOS_PUBLICACION_VALIDOS
    ):
        errores.append(
            "Tipo de publicacion no soportado."
        )

    datasets = manifest.get("datasets")

    if not isinstance(datasets, list):

        errores.append(
            "datasets debe ser una lista."
        )

        return errores, advertencias

    if not datasets:

        errores.append(
            "El manifest no contiene datasets."
        )

        return errores, advertencias

    datasets_vistos = set()

    for indice, dataset in enumerate(datasets):

        if isinstance(dataset, dict):

            nombre = dataset.get("dataset")

            if texto_valido(nombre):

                if nombre in datasets_vistos:
                    errores.append(
                        f"Dataset duplicado: {nombre}"
                    )

                datasets_vistos.add(nombre)

        validar_dataset(
            dataset,
            indice,
            entorno,
            errores,
            advertencias,
            config=config,
        )

    return errores, advertencias

# 13. CONSTRUIR RESULTADO
# ============================================================

def procesar_manifest(
    ruta,
    entorno_esperado="PRUEBA",
    backend=None,
):
    resultado = {
        "archivo": str(ruta),
        "run_id": None,
        "version_manifest": None,
        "entorno": None,
        "huella_sha256": None,
        "validacion_estructural_aprobada": False,
        "validacion_operativa_aprobada": False,
        "autorizacion_ejecucion_gold": False,
        "estado": "GOLD_RECHAZADO",
        "datasets": [],
        "errores": [],
        "advertencias": [],
    }

    try:
        if backend is None and str(ruta).startswith("gs://"):
            try:
                config = cargar_configuracion()
                if config.backend != BACKEND_GCS:
                    raise ManifestError(
                        "Se requiere configuracion GCS."
                    )
                backend = crear_backend(config)
            except Exception as exc:
                raise ManifestError(str(exc)) from exc

        manifest, huella = cargar_manifest(
            ruta,
            backend=backend,
        )

        resultado["huella_sha256"] = huella

        resultado["run_id"] = manifest.get(
            "run_id"
        )

        resultado["version_manifest"] = manifest.get(
            "version_manifest"
        )

        resultado["entorno"] = manifest.get(
            "entorno"
        )

        errores, advertencias = validar_manifest(
            manifest,
            entorno_esperado,
            config=backend.config if backend is not None else None,
        )

        resultado["errores"] = errores
        resultado["advertencias"] = advertencias

        datasets = manifest.get("datasets")

        if isinstance(datasets, list):

            resultado["datasets"] = [
                item.get("dataset")
                for item in datasets
                if isinstance(item, dict)
            ]

        if not errores:

            resultado[
                "validacion_estructural_aprobada"
            ] = True

            resultado["estado"] = (
                "MANIFEST_ESTRUCTURA_APROBADA"
            )

            resultado["advertencias"].append(
                "Validacion operativa de Silver pendiente. "
                "No se autoriza ejecutar Gold todavia."
            )

    except ManifestError as exc:

        resultado["errores"].append(str(exc))

    return resultado


# ============================================================
# 14. INTERFAZ DE LINEA DE COMANDOS
# ============================================================

def main():
    parser = argparse.ArgumentParser(
        description=(
            "Validador estructural de manifests "
            "Silver -> Gold para BVL Analytics."
        )
    )

    parser.add_argument(
        "--manifest",
        required=True,
        help="Ruta del manifest JSON.",
    )

    parser.add_argument(
        "--entorno",
        choices=["PRUEBA", "PRODUCCION"],
        default="PRUEBA",
        help="Entorno esperado. Default: PRUEBA.",
    )

    argumentos = parser.parse_args()

    resultado = procesar_manifest(
        argumentos.manifest,
        argumentos.entorno,
    )

    print(
        json.dumps(
            resultado,
            indent=4,
            ensure_ascii=False,
            allow_nan=False,
        )
    )

    if resultado[
        "validacion_estructural_aprobada"
    ]:
        return 0

    return 1


if __name__ == "__main__":
    sys.exit(main())

"""Vinculación de archivos al manifest Silver oficial autenticado."""

from validar_archivos_silver import validar_archivos_silver


class VinculacionManifestError(ValueError):
    pass


def validar_archivos_del_manifest(cliente, manifest, version_autorizada):
    if not isinstance(manifest, dict):
        raise VinculacionManifestError("Manifest inválido")

    if (
        not isinstance(version_autorizada, str)
        or not version_autorizada.strip()
        or manifest.get("version_id") != version_autorizada
    ):
        raise VinculacionManifestError("Versión incompatible")

    files = manifest.get("files")
    version_prefix = manifest.get("version_prefix")

    if not isinstance(files, list) or not files:
        raise VinculacionManifestError("Manifest sin archivos")

    if (
        not isinstance(version_prefix, str)
        or not version_prefix.endswith("/")
        or "/staging/" in version_prefix
    ):
        raise VinculacionManifestError("Prefijo inválido")

    resultado = validar_archivos_silver(
        cliente, files, version_prefix
    )

    return {
        "version_id": version_autorizada,
        "archivos_validados": resultado["archivos_validados"],
        "integridad_verificada": True,
        "consumo_productivo_autorizado": False,
    }
